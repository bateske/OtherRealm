cmake_minimum_required(VERSION 3.16)
project(Otherrealm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(otherrealm engine/aw_vm.cpp engine/aw_music.cpp engine/aw_pack.cpp engine/aw_render.cpp engine/aw_game.cpp engine/aw_checkpoint.cpp engine/aw_session.cpp)
target_include_directories(otherrealm PUBLIC engine)
add_executable(capture_title tools/capture_title.cpp engine/aw_vm.cpp engine/aw_music.cpp engine/aw_pack.cpp engine/aw_render.cpp engine/aw_game.cpp engine/aw_checkpoint.cpp engine/aw_session.cpp)
target_compile_definitions(capture_title PRIVATE OR_WIDTH=208)
add_executable(record_gameplay tools/record_gameplay.cpp)
target_link_libraries(record_gameplay PRIVATE otherrealm)
foreach(test headless vm_tests pack_tests render_tests demo_tests probe checkpoint_tests session_tests)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${test}.cpp")
  add_executable(${test} tests/${test}.cpp)
  target_link_libraries(${test} PRIVATE otherrealm)
 endif()
endforeach()
enable_testing()
add_executable(persistence_tests tests/persistence_tests.cpp firmware/Otherrealm/Persistence.cpp)
target_compile_definitions(persistence_tests PRIVATE OTHERREALM_PERSISTENCE_TEST)
add_test(NAME persistence COMMAND persistence_tests)
add_test(NAME checkpoint COMMAND checkpoint_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo/sd/OTHERWRL.PAK")
add_test(NAME session COMMAND session_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo/sd/OTHERWRL.PAK")
add_test(NAME vm COMMAND vm_tests)
add_test(NAME pack COMMAND pack_tests)
if(TARGET render_tests)
 add_test(NAME render COMMAND render_tests)
endif()
add_test(NAME demo COMMAND demo_tests "${CMAKE_CURRENT_SOURCE_DIR}/demo/sd/OTHERWRL.PAK")
